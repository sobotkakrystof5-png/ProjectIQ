import { NextRequest, NextResponse } from 'next/server'
import { readFormDataLimited } from '@/lib/security/request-body'
import { categoryFor } from '@/lib/admin/voice/categories'
import { cleanupWithClaude } from '@/lib/admin/voice/cleanup'
import { authorizeVoiceDevice, dictationMetaSchema, startedAtPlausible, voiceFail } from '@/lib/admin/voice/http'
import { runPipeline } from '@/lib/admin/voice/pipeline'
import { findDictation, loadPipelineContext, saveCancelled, saveDictation } from '@/lib/admin/voice/store'
import { transcribeVoice } from '@/lib/admin/voice/stt'

// ZakazIQ Voice (R-150): diktát z Macu → text.
//
// Multipart: `audio` (m4a/wav, nejvýš 4 MB ≈ 20 minut AAC 16 kHz mono) a `meta`
// (JSON podle dictationMetaSchema). Server přepíše, projde pojistkami proti
// halucinacím, učeše a vrátí text; zvuk nikam neukládá. Stejné `id` podruhé
// vrátí uložený výsledek bez nového přepisu (fronta offline na Macu).

export const maxDuration = 60
export const dynamic = 'force-dynamic'

const MAX_AUDIO_BYTES = 4 * 1024 * 1024
const MAX_BODY_BYTES = MAX_AUDIO_BYTES + 64 * 1024
const AUDIO_TYPES = new Set(['audio/mp4', 'audio/x-m4a', 'audio/m4a', 'audio/aac', 'audio/wav', 'audio/x-wav', 'audio/webm', 'audio/mpeg'])

export async function POST(req: NextRequest): Promise<NextResponse> {
  const started = Date.now()
  const auth = await authorizeVoiceDevice(req, 'dictate')
  if (!auth.ok) return auth.res

  const form = await readFormDataLimited(req, MAX_BODY_BYTES)
  if (!form.ok) return form.reason === 'too-large' ? voiceFail(413, 'TOO_LARGE') : voiceFail(400, 'INVALID_BODY')

  let metaJson: unknown
  try {
    const raw = form.form.get('meta')
    metaJson = typeof raw === 'string' ? JSON.parse(raw) : null
  } catch {
    return voiceFail(400, 'INVALID_META')
  }
  const parsed = dictationMetaSchema.safeParse(metaJson)
  if (!parsed.success || !startedAtPlausible(parsed.data.startedAt)) return voiceFail(400, 'INVALID_META')
  const m = parsed.data

  const category = categoryFor(m.app.bundleId, m.app.titleHint ?? null)
  const meta = {
    id: m.id,
    deviceId: auth.device.id,
    startedAt: m.startedAt,
    durationMs: m.durationMs,
    speechMs: m.speechMs,
    mode: m.mode,
    appBundleId: m.app.bundleId,
    appName: m.app.name,
    category,
  }

  const existing = await findDictation(m.id, auth.device.id)
  if (existing) {
    return NextResponse.json({ ok: true, id: m.id, status: existing.status, text: existing.finalText ?? '', words: existing.words, guards: existing.guards })
  }

  if (m.cancelled) {
    await saveCancelled(meta)
    return NextResponse.json({ ok: true, id: m.id, status: 'cancelled', text: '', words: 0, guards: [] })
  }

  const audio = form.form.get('audio')
  if (!(audio instanceof Blob) || audio.size === 0) return voiceFail(400, 'INVALID_AUDIO')
  if (audio.size > MAX_AUDIO_BYTES) return voiceFail(413, 'TOO_LARGE')
  if (!AUDIO_TYPES.has(audio.type.split(';')[0]?.trim().toLowerCase() ?? '')) return voiceFail(415, 'UNSUPPORTED_TYPE')

  const { config } = await loadPipelineContext(category)
  const outcome = await runPipeline(
    { audio, durationMs: m.durationMs, speechMs: m.speechMs, category, appName: m.app.name },
    config,
    { transcribe: transcribeVoice, cleanup: cleanupWithClaude },
  )
  await saveDictation(meta, outcome, Date.now() - started)

  if (outcome.status === 'error') {
    const code = outcome.guards.includes('stt_not_configured') ? 'STT_NOT_CONFIGURED' : 'STT_FAILED'
    return voiceFail(502, code)
  }
  return NextResponse.json({
    ok: true,
    id: m.id,
    status: outcome.status,
    text: outcome.finalText,
    words: outcome.words,
    guards: outcome.guards,
  })
}
